package main

import (
	"context"
	"fmt"
	"time"
)

func main() {

	// 1
	ctx, cancel := context.WithCancel(context.Background())

	go func() {
		time.Sleep(2 * time.Second)
		cancel()
	}()

	fmt.Println("start")
	<-ctx.Done()
	fmt.Println("end")

	//2
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()

	fmt.Println("start")
	<-ctx.Done()
	fmt.Println("end")
}
