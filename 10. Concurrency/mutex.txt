package main

import (
	"fmt"
	"sync"
	"time"
)

// | Function         | Type      | Purpose             | Readers Allowed | Writers Allowed |
// | ---------------- | --------- | ------------------- | --------------- | --------------- |
// | `mu.Lock()`      | `Mutex`   | Full exclusive lock | 0               | 1               |
// | `mu.Unlock()`    | `Mutex`   | Release lock        | —               | —               |
// | `rwmu.RLock()`   | `RWMutex` | Read-only lock      | Multiple        | 0               |
// | `rwmu.RUnlock()` | `RWMutex` | Release read lock   | —               | —               |
// | `rwmu.Lock()`    | `RWMutex` | Write lock          | 0               | 1               |
// | `rwmu.Unlock()`  | `RWMutex` | Release write lock  | —               | —               |

var counter int
var mu sync.Mutex

func main() {

	for range 3 {
		go func() {
			mu.Lock()
			counter++
			mu.Unlock()
		}()
	}

	time.Sleep(time.Second)

	fmt.Println("Counter:", counter)
}
