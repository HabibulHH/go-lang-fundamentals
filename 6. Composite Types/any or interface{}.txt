	// any / interface{}

	var value any

	value = 10
	fmt.Println(value)

	value = true
	fmt.Println(value)

	value = "hello"
	fmt.Println(value)

	data := []any{10, true, "hello", 3.14}
	fmt.Println(data)

	user := map[string]any{
		"name":   "Hira",
		"age":    30,
		"height": 5.10,
		"skills": []string{"Go", "Rust", "JavaScript"},
	}
	fmt.Println(user)
